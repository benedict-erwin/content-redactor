# === Domains ===
production.example.com=internal.example.local
staging.example.com=internal.staging.local
api.live.example.com=api.internal.local

# === IP Addresses ===

172.16.5.100=10.0.0.100
203.0.113.50=10.0.0.50

# === Emails ===
admin@realcompany.com=admin@internal.local
user@production.example.com=user@internal.local
support@staging.example.com=support@internal.local

# === Usernames ===
admin-prod=admin-dev
superadmin=internal-admin
dev-user=test-user

# === URLs (best used with --regex) ===
https?://api\.production\.example\.com=https://api.internal.local
https?://app\.staging\.example\.com=https://app.internal.local

# === Tokens & Secrets ===
sk_live_[a-zA-Z0-9]{20,}=REDACTED_SECRET_KEY
Bearer [a-zA-Z0-9]{30,}=REDACTED_BEARER_TOKEN

# === Generic Sensitive Words ===

secret=REDACTED_SECRET
api_key=REDACTED_API_KEY
private-key=REDACTED_PRIVATE_KEY

# === Environment Flags ===
PROD_ENV=INTERNAL_ENV

STAGING=false
